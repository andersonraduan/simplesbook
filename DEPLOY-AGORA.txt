cd ~/calendario.clinicalasante.pt && \
source /home/clini489/nodevenv/calendario.clinicalasante.pt/24/bin/activate && \

chmod -R 755 .next && \

sleep 5 && \
echo "=== Status ===" && \
pm2 status && \
echo "" && \
echo "=== Logs ===" && \
pm2 logs simplesbook --lines 20 --nostream
