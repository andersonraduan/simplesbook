COMANDO FINAL - USAR NEXT START DIRETAMENTE
===========================================

IMPORTANTE: Envie o arquivo ecosystem.config.js atualizado via FTP primeiro!

Depois execute no servidor:

cd ~/calendario.clinicalasante.pt && \
export PATH="$HOME/nodevenv/calendario.clinicalasante.pt/24/bin:$PATH" && \
echo "=== 1. Parando PM2 ===" && \
pm2 stop simplesbook 2>/dev/null || true && \
pm2 delete simplesbook 2>/dev/null || true && \
echo "✅ PM2 parado" && \
echo "" && \
echo "=== 2. Verificando se ecosystem.config.js foi atualizado ===" && \
if grep -q "next/dist/bin/next" ecosystem.config.js 2>/dev/null; then \
    echo "✅ ecosystem.config.js atualizado"; \
else \
    echo "⚠️ ecosystem.config.js NÃO foi atualizado! Envie o arquivo via FTP primeiro."; \
    exit 1; \
fi && \
echo "" && \
echo "=== 3. Removendo TODOS os caches do Next.js ===" && \
rm -rf .next/cache .next/.turbo .next/trace .next/dev 2>/dev/null || true && \
echo "✅ Caches removidos" && \
echo "" && \
echo "=== 4. Verificando se build existe ===" && \
if [ ! -d ".next/server" ]; then \
    echo "❌ Build não existe! Você precisa fazer build primeiro."; \
    exit 1; \
fi && \
echo "✅ Build existe" && \
echo "" && \
echo "=== 5. Iniciando PM2 com next start (não usa Turbopack) ===" && \
NODE_ENV=production NEXT_TURBOPACK_DISABLE=1 NEXT_PRIVATE_DISABLE_TURBOPACK=1 pm2 start ecosystem.config.js --update-env && \
pm2 save && \
echo "✅ PM2 iniciado" && \
echo "" && \
echo "=== 6. Aguardando 30 segundos ===" && \
sleep 30 && \
echo "" && \
echo "=== 7. Verificando status ===" && \
pm2 status && \
echo "" && \
echo "=== 8. Verificando logs (não deve ter erros de Turbopack) ===" && \
pm2 logs simplesbook --lines 20 --nostream 2>&1 | tail -20 && \
echo "" && \
echo "=== 9. Verificando porta 3000 ===" && \
if netstat -tuln 2>/dev/null | grep -q ":3000"; then \
    echo "✅ Porta 3000 está escutando!" && \
    netstat -tuln | grep 3000; \
elif lsof -i :3000 2>/dev/null | grep -q LISTEN; then \
    echo "✅ Porta 3000 está escutando!" && \
    lsof -i :3000 | grep LISTEN; \
else \
    echo "❌ Porta 3000 ainda não está escutando" && \
    echo "Últimos logs de erro:" && \
    pm2 logs simplesbook --err --lines 30 --nostream 2>&1 | tail -30; \
fi && \
echo "" && \
echo "=== 10. Testando conexão ===" && \
if curl -s -o /dev/null -w "%{http_code}" http://localhost:3000 2>/dev/null | grep -q "200\|301\|302"; then \
    echo "✅ Aplicação responde!" && \
    curl -I http://localhost:3000 2>&1 | head -3; \
else \
    echo "❌ Aplicação ainda não responde" && \
    echo "Ver logs: pm2 logs simplesbook --err"; \
fi && \
echo "" && \
echo "✅ PROCESSO CONCLUÍDO!"
