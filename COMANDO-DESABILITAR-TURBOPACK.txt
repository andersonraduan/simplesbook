COMANDO PARA DESABILITAR TURBOPACK E CORRIGIR ERRO
===================================================

IMPORTANTE: Primeiro, envie os arquivos atualizados via FTP:
- next.config.js
- server.js

Depois, execute este comando no servidor:

cd ~/calendario.clinicalasante.pt && \
export PATH="$HOME/nodevenv/calendario.clinicalasante.pt/24/bin:$PATH" && \
echo "=== 1. Parando PM2 ===" && \
pm2 stop simplesbook 2>/dev/null || true && \
pm2 delete simplesbook 2>/dev/null || true && \
echo "✅ PM2 parado" && \
echo "" && \
echo "=== 2. Removendo cache do Next.js ===" && \
rm -rf .next/cache .next/.turbo .next/trace 2>/dev/null || true && \
echo "✅ Cache removido" && \
echo "" && \
echo "=== 3. Verificando se arquivos foram atualizados ===" && \
if grep -q "NEXT_TURBOPACK_DISABLE" server.js 2>/dev/null; then \
    echo "✅ server.js atualizado"; \
else \
    echo "⚠️ server.js NÃO foi atualizado! Envie o arquivo via FTP primeiro."; \
    exit 1; \
fi && \
if grep -q "turbopack: false" next.config.js 2>/dev/null || grep -q "NEXT_TURBOPACK_DISABLE" next.config.js 2>/dev/null; then \
    echo "✅ next.config.js atualizado"; \
else \
    echo "⚠️ next.config.js NÃO foi atualizado! Envie o arquivo via FTP primeiro."; \
    exit 1; \
fi && \
echo "" && \
echo "=== 4. Configurando variáveis de ambiente ===" && \
export NODE_ENV=production && \
export NEXT_TURBOPACK_DISABLE=1 && \
export NEXT_PRIVATE_DISABLE_TURBOPACK=1 && \
echo "✅ Variáveis configuradas" && \
echo "" && \
echo "=== 5. Testando server.js manualmente (5 segundos) ===" && \
timeout 5 NODE_ENV=production NEXT_TURBOPACK_DISABLE=1 NEXT_PRIVATE_DISABLE_TURBOPACK=1 node server.js 2>&1 & \
TEST_PID=$! && \
sleep 3 && \
if ps -p $TEST_PID > /dev/null 2>&1; then \
    echo "✅ server.js iniciou sem erros de Turbopack" && \
    kill $TEST_PID 2>/dev/null || true; \
else \
    echo "⚠️ server.js pode ter problemas. Verificando logs..." && \
    sleep 2; \
fi && \
echo "" && \
echo "=== 6. Iniciando PM2 com variáveis ===" && \
NODE_ENV=production NEXT_TURBOPACK_DISABLE=1 NEXT_PRIVATE_DISABLE_TURBOPACK=1 pm2 start ecosystem.config.js --update-env && \
pm2 save && \
echo "✅ PM2 iniciado" && \
echo "" && \
echo "=== 7. Aguardando 25 segundos ===" && \
sleep 25 && \
echo "" && \
echo "=== 8. Verificando status ===" && \
pm2 status && \
echo "" && \
echo "=== 9. Verificando logs (não deve ter erros de Turbopack) ===" && \
pm2 logs simplesbook --lines 15 --nostream 2>&1 | tail -15 && \
echo "" && \
echo "=== 10. Verificando porta 3000 ===" && \
if netstat -tuln 2>/dev/null | grep -q ":3000"; then \
    echo "✅ Porta 3000 está escutando!" && \
    netstat -tuln | grep 3000; \
elif lsof -i :3000 2>/dev/null | grep -q LISTEN; then \
    echo "✅ Porta 3000 está escutando!" && \
    lsof -i :3000 | grep LISTEN; \
else \
    echo "❌ Porta 3000 ainda não está escutando" && \
    echo "Últimos logs de erro:" && \
    pm2 logs simplesbook --err --lines 20 --nostream 2>&1 | tail -20; \
fi && \
echo "" && \
echo "=== 11. Testando conexão ===" && \
if curl -s -o /dev/null -w "%{http_code}" http://localhost:3000 2>/dev/null | grep -q "200\|301\|302"; then \
    echo "✅ Aplicação responde!" && \
    curl -I http://localhost:3000 2>&1 | head -3; \
else \
    echo "❌ Aplicação ainda não responde" && \
    echo "Ver logs: pm2 logs simplesbook --err"; \
fi && \
echo "" && \
echo "✅ PROCESSO CONCLUÍDO!"
